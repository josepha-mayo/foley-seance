/**
 * @typedef {Object} AuditionSettings
 * @property {'creak'|'breath'} kind
 * @property {24000|48000} sampleRate
 * @property {number} x - 0..1
 * @property {number} y - 0..1
 * @property {boolean} doorClosed
 */

/**
 * @typedef {Object} AuditionResult
 * @property {Uint8Array} wavBytes
 * @property {number} sampleRate
 * @property {number} frames
 * @property {'creak'|'breath'} kind
 */

import { createDemoSource, renderRecipe, encodeWav } from './foley-engine.mjs';

const FADE_MS = 20;
const MASTER_GAIN = 0.35;
const MAX_NORMALIZE_THRESHOLD = 0.99; // Reject if renderer normalized beyond expected

/**
 * Build a looping audition with static placement.
 * @param {AuditionSettings} settings
 * @returns {AuditionResult}
 */
export function buildAudition(settings) {
  // Validate exact plain data
  if (!settings || typeof settings !== 'object') {
    throw new Error('Settings must be a plain object');
  }

  const { kind, sampleRate, x, y, doorClosed } = settings;

  if (kind !== 'creak' && kind !== 'breath') {
    throw new Error(`Invalid kind: ${kind}`);
  }

  if (sampleRate !== 24000 && sampleRate !== 48000) {
    throw new Error(`Invalid sampleRate: ${sampleRate}`);
  }

  if (typeof x !== 'number' || x < 0 || x > 1 || !Number.isFinite(x)) {
    throw new Error(`Invalid x: ${x}`);
  }

  if (typeof y !== 'number' || y < 0 || y > 1 || !Number.isFinite(y)) {
    throw new Error(`Invalid y: ${y}`);
  }

  if (typeof doorClosed !== 'boolean') {
    throw new Error(`Invalid doorClosed: ${doorClosed}`);
  }

  // Create source (<=8sec mono)
  const source = createDemoSource(sampleRate, kind);

  // Build static recipe
  const sourceDuration = source.length / sampleRate;
  const recipe = {
    version: 1,
    doorClosed,
    trajectory: [
      { t: 0, x, y },
      { t: sourceDuration, x, y }
    ]
  };

  // Render using shared artistic room DSP
  const renderResult = renderRecipe(source, sampleRate, recipe);

  // Validate render result
  if (!renderResult || typeof renderResult !== 'object') {
    throw new Error('Invalid render result');
  }

  const { wavBytes, previewChannels, outputFrames, sourceFrames, tailFrames, normalizationGain } = renderResult;

  if (!Array.isArray(previewChannels) || previewChannels.length !== 2) {
    throw new Error('Expected stereo preview channels');
  }

  if (!(previewChannels[0] instanceof Float32Array) || !(previewChannels[1] instanceof Float32Array)) {
    throw new Error('Preview channels must be Float32Array');
  }

  if (previewChannels[0].length !== outputFrames || previewChannels[1].length !== outputFrames) {
    throw new Error('Channel length mismatch');
  }

  // Validate finite samples
  for (let ch = 0; ch < 2; ch++) {
    const channel = previewChannels[ch];
    for (let i = 0; i < channel.length; i++) {
      if (!Number.isFinite(channel[i])) {
        throw new Error(`Non-finite sample at channel ${ch}, frame ${i}`);
      }
    }
  }

  // Check normalization - reject unexpected gain reduction
  if (typeof normalizationGain === 'number' && normalizationGain < MAX_NORMALIZE_THRESHOLD) {
    throw new Error(`Unexpected normalization gain: ${normalizationGain}`);
  }

  // Apply master audition gain and loop boundary fade
  const fadeSamples = Math.floor((FADE_MS / 1000) * sampleRate);
  const fadedChannels = [
    new Float32Array(outputFrames),
    new Float32Array(outputFrames)
  ];

  for (let ch = 0; ch < 2; ch++) {
    const src = previewChannels[ch];
    const dst = fadedChannels[ch];

    for (let i = 0; i < outputFrames; i++) {
      let gain = MASTER_GAIN;

      // Fade in at start
      if (i < fadeSamples) {
        gain *= i / fadeSamples;
      }

      // Fade out at end
      if (i >= outputFrames - fadeSamples) {
        gain *= (outputFrames - i) / fadeSamples;
      }

      dst[i] = src[i] * gain;
    }
  }

  // Encode edited channels to exact WAV
  const finalWavBytes = encodeWav(fadedChannels, sampleRate);

  return {
    wavBytes: finalWavBytes,
    sampleRate,
    frames: outputFrames,
    kind
  };
}

/**
 * @typedef {Object} AuditionControllerConfig
 * @property {AudioContext} context
 * @property {Worker} worker
 * @property {(bytes: Uint8Array) => Promise<AudioBuffer>} decode
 * @property {(state: AuditionState) => void} onState
 * @property {(error: Error) => void} onError
 * @property {() => number} [now]
 */

/**
 * @typedef {Object} AuditionState
 * @property {'stopped'|'pending'|'playing'} state
 * @property {number} revision
 * @property {number} [latencyMs]
 */

const CROSSFADE_MS = 40;

/**
 * Create an audition controller for looping spatial sketches.
 * @param {AuditionControllerConfig} config
 */
export function createAuditionController(config) {
  const { context, worker, decode, onState, onError, now = () => performance.now() } = config;

  let revision = 0;
  let currentEpoch = null; // { revision, kind, sampleRate, buffer, startTime, sourceNode, gainNode }
  let pendingRequest = null; // { revision, settings, startTime }
  let inflightRequest = null; // { revision, settings, startTime }
  let retiringNodes = []; // { sourceNode, gainNode, endTime }

  function emitState(state, latencyMs) {
    onState({
      state,
      revision,
      latencyMs: latencyMs !== undefined ? latencyMs : undefined
    });
  }

  function cleanup() {
    // Stop and disconnect all active and retiring nodes
    if (currentEpoch) {
      stopNode(currentEpoch.sourceNode, currentEpoch.gainNode);
      currentEpoch = null;
    }

    for (const retiring of retiringNodes) {
      stopNode(retiring.sourceNode, retiring.gainNode);
    }
    retiringNodes = [];

    pendingRequest = null;
    inflightRequest = null;
  }

  function stopNode(sourceNode, gainNode) {
    try {
      sourceNode.stop();
    } catch (e) {
      // Already stopped
    }
    try {
      sourceNode.disconnect();
    } catch (e) {
      // Already disconnected
    }
    try {
      gainNode.disconnect();
    } catch (e) {
      // Already disconnected
    }
  }

  function processQueue() {
    // If worker busy or nothing pending, return
    if (inflightRequest || !pendingRequest) {
      return;
    }

    // Move pending to inflight
    inflightRequest = pendingRequest;
    pendingRequest = null;

    const request = inflightRequest;

    worker.postMessage({
      id: request.revision,
      settings: request.settings
    });

    emitState('pending');
  }

  worker.onmessage = async function(event) {
    const { id, result, error } = event.data;

    // Check if stale
    if (id !== revision) {
      return; // Discard stale response
    }

    if (!inflightRequest || inflightRequest.revision !== id) {
      return; // Not our request
    }

    const request = inflightRequest;
    inflightRequest = null;

    if (error) {
      try {
        onError(new Error(`Worker error: ${error}`));
      } catch (e) {
        // Ignore onError exceptions
      }
      emitState('stopped');
      processQueue();
      return;
    }

    // Validate result
    if (!result || typeof result !== 'object') {
      try {
        onError(new Error('Invalid worker result'));
      } catch (e) {}
      emitState('stopped');
      processQueue();
      return;
    }

    const { wavBytes, sampleRate, frames, kind } = result;

    if (!(wavBytes instanceof Uint8Array)) {
      try {
        onError(new Error('wavBytes must be Uint8Array'));
      } catch (e) {}
      emitState('stopped');
      processQueue();
      return;
    }

    let buffer;
    try {
      buffer = await decode(wavBytes);
    } catch (decodeError) {
      // Check if stale after decode
      if (id !== revision) {
        processQueue();
        return;
      }

      try {
        onError(new Error(`Decode failed: ${decodeError.message}`));
      } catch (e) {}
      emitState('stopped');
      processQueue();
      return;
    }

    // Check if stale after async decode
    if (id !== revision) {
      processQueue();
      return;
    }

    // Validate decoded buffer
    if (buffer.numberOfChannels !== 2) {
      try {
        onError(new Error('Expected stereo buffer'));
      } catch (e) {}
      emitState('stopped');
      processQueue();
      return;
    }

    if (buffer.sampleRate !== sampleRate) {
      try {
        onError(new Error(`Sample rate mismatch: ${buffer.sampleRate} vs ${sampleRate}`));
      } catch (e) {}
      emitState('stopped');
      processQueue();
      return;
    }

    if (buffer.length !== frames) {
      try {
        onError(new Error(`Frame count mismatch: ${buffer.length} vs ${frames}`));
      } catch (e) {}
      emitState('stopped');
      processQueue();
      return;
    }

    // Validate finite peak
    for (let ch = 0; ch < 2; ch++) {
      const channelData = buffer.getChannelData(ch);
      for (let i = 0; i < channelData.length; i++) {
        if (!Number.isFinite(channelData[i])) {
          try {
            onError(new Error(`Non-finite sample in decoded buffer at channel ${ch}, frame ${i}`));
          } catch (e) {}
          emitState('stopped');
          processQueue();
          return;
        }
      }
    }

    const latencyMs = now() - request.startTime;

    // Create new epoch or update existing
    const contextTime = context.currentTime;
    const bufferDuration = buffer.duration;
    const crossfadeDuration = CROSSFADE_MS / 1000;

    if (!currentEpoch || currentEpoch.kind !== kind || currentEpoch.sampleRate !== sampleRate) {
      // Source change: start new epoch
      if (currentEpoch) {
        // Retire old epoch
        const oldSource = currentEpoch.sourceNode;
        const oldGain = currentEpoch.gainNode;

        // Fade out
        oldGain.gain.cancelScheduledValues(contextTime);
        oldGain.gain.setValueAtTime(oldGain.gain.value, contextTime);
        oldGain.gain.linearRampToValueAtTime(0, contextTime + crossfadeDuration);

        // Schedule stop and disconnect at audio-time end
        const endTime = contextTime + crossfadeDuration;
        retiringNodes.push({ sourceNode: oldSource, gainNode: oldGain, endTime });

        oldSource.onended = () => {
          stopNode(oldSource, oldGain);
          retiringNodes = retiringNodes.filter(n => n.sourceNode !== oldSource);
        };

        try {
          oldSource.stop(endTime);
        } catch (e) {
          stopNode(oldSource, oldGain);
          retiringNodes = retiringNodes.filter(n => n.sourceNode !== oldSource);
        }
      }

      // Create new epoch
      const sourceNode = context.createBufferSource();
      sourceNode.buffer = buffer;
      sourceNode.loop = true;

      const gainNode = context.createGain();
      gainNode.gain.setValueAtTime(0, contextTime);
      gainNode.gain.linearRampToValueAtTime(1, contextTime + crossfadeDuration);

      sourceNode.connect(gainNode);
      gainNode.connect(context.destination);

      try {
        sourceNode.start(contextTime);
      } catch (startError) {
        stopNode(sourceNode, gainNode);
        try {
          onError(new Error(`Failed to start source: ${startError.message}`));
        } catch (e) {}
        emitState('stopped');
        processQueue();
        return;
      }

      currentEpoch = {
        revision: id,
        kind,
        sampleRate,
        buffer,
        startTime: contextTime,
        sourceNode,
        gainNode
      };

      emitState('playing', latencyMs);
    } else {
      // Same source: crossfade to new buffer with phase alignment
      const oldSource = currentEpoch.sourceNode;
      const oldGain = currentEpoch.gainNode;
      const epochStartTime = currentEpoch.startTime;

      // Calculate phase-aligned start offset
      const elapsedSinceEpoch = contextTime - epochStartTime;
      const offset = elapsedSinceEpoch % bufferDuration;

      // Create new source at aligned offset
      const newSource = context.createBufferSource();
      newSource.buffer = buffer;
      newSource.loop = true;

      const newGain = context.createGain();
      newGain.gain.setValueAtTime(0, contextTime);
      newGain.gain.linearRampToValueAtTime(1, contextTime + crossfadeDuration);

      newSource.connect(newGain);
      newGain.connect(context.destination);

      // Complementary fade out on old
      oldGain.gain.cancelScheduledValues(contextTime);
      oldGain.gain.setValueAtTime(oldGain.gain.value, contextTime);
      oldGain.gain.linearRampToValueAtTime(0, contextTime + crossfadeDuration);

      try {
        newSource.start(contextTime, offset);
      } catch (startError) {
        stopNode(newSource, newGain);
        try {
          onError(new Error(`Failed to start source: ${startError.message}`));
        } catch (e) {}
        emitState('stopped');
        processQueue();
        return;
      }

      // Retire old source
      const endTime = contextTime + crossfadeDuration;
      retiringNodes.push({ sourceNode: oldSource, gainNode: oldGain, endTime });

      oldSource.onended = () => {
        stopNode(oldSource, oldGain);
        retiringNodes = retiringNodes.filter(n => n.sourceNode !== oldSource);
      };

      try {
        oldSource.stop(endTime);
      } catch (e) {
        stopNode(oldSource, oldGain);
        retiringNodes = retiringNodes.filter(n => n.sourceNode !== oldSource);
      }

      // Update epoch
      currentEpoch = {
        revision: id,
        kind,
        sampleRate,
        buffer,
        startTime: epochStartTime, // Preserve epoch start
        sourceNode: newSource,
        gainNode: newGain
      };

      emitState('playing', latencyMs);
    }

    // Process next pending
    processQueue();
  };

  worker.onerror = function(event) {
    try {
      onError(new Error(`Worker error: ${event.message}`));
    } catch (e) {}
    
    if (inflightRequest && inflightRequest.revision === revision) {
      inflightRequest = null;
      emitState('stopped');
      processQueue();
    }
  };

  return {
    start(settings) {
      revision++;
      cleanup();

      pendingRequest = {
        revision,
        settings,
        startTime: now()
      };

      processQueue();
    },

    update(settings) {
      if (!currentEpoch && !inflightRequest && !pendingRequest) {
        // No active state, treat as start
        this.start(settings);
        return;
      }

      // Check for source change
      if (currentEpoch && (settings.kind !== currentEpoch.kind || settings.sampleRate !== currentEpoch.sampleRate)) {
        // Treat as new start
        this.start(settings);
        return;
      }

      revision++;

      // Replace pending
      pendingRequest = {
        revision,
        settings,
        startTime: now()
      };

      processQueue();
    },

    stop() {
      revision++;
      cleanup();
      emitState('stopped');
    },

    dispose() {
      revision++;
      cleanup();
      worker.terminate();
    }
  };
}

# Audition Module Contract

## Purpose
Minimal dependency-free ESM module for bounded static audition in Foley Seance browser-local sound room. Provides user-started looping spatial sketches with synthetic creak/breath sources, static placement (x, y), and door state toggle only.

## Scope and Boundaries

### In Scope
- User-initiated looping audition with explicit start
- Static spatial placement (x, y coordinates 0..1)
- Door state (open/closed) toggle
- Synthetic builtin sources: creak and breath (<=8sec mono)
- Fixed sample rates: 24000 or 48000 Hz
- Bounded loop duration (full source + fixed 1.5sec tail)
- Phase-aligned crossfade updates during playback
- Worker-based rendering with transferable buffers

### Out of Scope
- Authored movement/render/export (separate, untouched workflows)
- Recording performances or trajectory capture
- Autoplay, background audio, or automatic resume
- Microphone input or live audio processing
- User accounts, persistence, or cloud features
- Custom UI components (parent integrates)
- Real-time guarantees (best-effort latency reporting)
- Dynamic source upload or different sound graph
- Continuous room physics simulation claims

## Architecture

### Core (`audition-core.mjs`)
Pure function `buildAudition(settings)` that:
1. Validates exact plain data schema
2. Creates demo source via shared `createDemoSource(rate, kind)`
3. Builds static recipe with fixed x/y trajectory
4. Calls shared `renderRecipe(source, rate, recipe)` containing artistic room DSP
5. Applies fixed master gain (0.35) and 20ms loop boundary fades
6. Encodes edited PCM to WAV via shared `encodeWav(channels, rate)`
7. Returns `{wavBytes, sampleRate, frames, kind}`

**Normalization Convention**: Shared renderer applies down-only normalization per buffer. For these builtin sources, normalization should not trigger. Module rejects unexpected gain reduction (< 0.99) rather than accepting gain pumping. Fixed master gain 0.35 applied by controller, never automatic loudness compensation.

**Immutability**: Does not mutate shared engine output; creates new Float32Array for faded channels.

### Worker (`audition-worker.mjs`)
- Imports core
- Handles `{id, settings}` messages
- Returns `{id, result}` or `{id, error}`
- Transfers `wavBytes.buffer` for efficiency
- Guarded `self` check for safe node import

### Controller (`audition-controller.mjs`)
Factory `createAuditionController(config)` with:

**Config Requirements**:
- `context`: AudioContext (caller creates and resumes on explicit user gesture)
- `worker`: Web Worker instance
- `decode(bytes)`: Promise<AudioBuffer> injected for native decoding
- `onState(state)`: Receives `{state, revision, latencyMs?}`
- `onError(error)`: Error callback
- `now()`: Optional, defaults to `performance.now`

**API**:
- `start(settings)`: Begin audition from explicit caller interaction only
- `update(settings)`: Update parameters during playback
- `stop()`: Stop immediately, invalidate epoch
- `dispose()`: Cleanup and terminate worker

**State Management**:
- Single inflight worker request + newest pending (stale requests discarded)
- Stale revision checks after both worker response AND async decode
- Never starve latest while dragging; pending replaces previous pending
- Active kind/sampleRate change treated as stop + explicit new start

**Audio Graph**:
- AudioBufferSource (looping) → GainNode → destination
- Phase alignment: offset calculated from shared audio context start epoch
- Linear complementary 40ms crossfade ramps (not equal-power, signals correlated)
- Retiring nodes stopped/disconnected at audio-time end via `onended` callback
- No setTimeout for sound timing guarantees
- Bounded total gain: serialize transitions safely

**Safety**:
- Stop/dispose prevents stale decode results from restarting audio
- Handles worker errors, decode rejection, source start exception
- No unhandled promise rejections
- Validates decoded stereo/rate/framecount and finite peaks before playback

**Limitations**:
- Updates show pending state and measured latency only
- No real-time guarantees (best-effort async rendering)
- No browser reads or hidden global side effects at import time

## Ownership and Responsibilities

### Caller Responsibilities
- Create and own AudioContext lifecycle
- Call `context.resume()` on explicit user gesture (e.g., click handler)
- Provide Web Worker instance
- Provide native decode function (e.g., `context.decodeAudioData.bind(context)`)
- Handle state changes via `onState` callback
- Handle errors via `onError` callback
- Call `start()` only from explicit user interaction
- Integrate into UI (module provides no UI components)

### Module Responsibilities
- Validate input settings schema and reject malformed data
- Render audio via worker without blocking main thread
- Manage audio graph lifecycle (source, gain, crossfades)
- Prevent stale audio playback after stop/dispose
- Report state transitions and measured latency
- Clean up all resources on dispose

### Shared Engine Responsibilities
- Provide `createDemoSource(rate, kind)` for synthetic sources
- Provide `renderRecipe(source, rate, recipe)` with artistic room DSP
- Provide `encodeWav(channels, rate)` for WAV encoding
- Apply fixed 1.5sec tail processing
- Return normalized stereo preview channels and metadata

## Testing Requirements
Tests use `node:test` mocks, no external dependencies:
- Settings validation (exact schema enforcement)
- Stop during decode prevents playback
- Queue burst with single inflight limit
- Stale revision discard after new start
- Phase alignment verification (shared epoch preserved)
- Complementary ramps and bounded node count
- Error handling: worker error, decode rejection, malformed response
- Worker core settings validation

## Performance Characteristics
- Worker rendering: non-blocking main thread
- Transferable buffers: zero-copy worker → main
- Measured latency: worker + decode time reported
- Fixed loop duration: full source + 1.5sec tail
- Crossfade overhead: 40ms transition period
- Memory: bounded by single active buffer + short retiring queue

## Limitations and Non-Goals
- No autoplay or background audio (explicit user start required)
- No real-time hard guarantees (async rendering, best-effort latency)
- No trajectory recording or authored performance capture
- No dynamic source upload or arbitrary audio files
- No microphone input or live audio processing
- No persistence, accounts, or cloud synchronization
- No UI components (parent integration responsibility)
- No claims of continuous room physics simulation
- Builtin sources only (creak, breath) - no custom sound graph
