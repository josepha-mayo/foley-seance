```javascript
/**
 * custom-motion.mjs
 * Pure browser JavaScript ES module for Foley Seance motion trajectory normalization.
 * No imports, DOM, IO, network, audio APIs, or global mutations.
 */

/**
 * Validates and normalizes motion samples to a specified duration.
 * @param {Array} samples - Array of 2-128 motion points with {x, y, t} properties
 * @param {number} duration - Target duration in seconds (>0, <=8)
 * @returns {Array} Normalized array with t mapped to [0, duration]
 */
export function normalizeMotion(samples, duration = 1) {
  // Validate duration
  if (typeof duration !== 'number' || !Number.isFinite(duration) || duration <= 0 || duration > 8) {
    throw new Error('Duration must be a finite number >0 and <=8');
  }

  // Validate array shape
  if (!Array.isArray(samples) || Object.getPrototypeOf(samples) !== Array.prototype) {
    throw new Error('Samples must be a plain Array');
  }
  if (samples.length < 2 || samples.length > 128) {
    throw new Error('Samples must contain 2-128 points');
  }

  // Check for dense array (no holes, no extra properties)
  const desc = Object.getOwnPropertyDescriptors(samples);
  for (let i = 0; i < samples.length; i++) {
    if (!(i in desc) || desc[i].get || desc[i].set) {
      throw new Error('Array must be dense with no accessor properties');
    }
  }
  for (const key of Object.keys(samples)) {
    if (key !== 'length' && (Number(key) < 0 || Number(key) >= samples.length || !Number.isInteger(Number(key)))) {
      throw new Error('Array must have only numeric indices and length');
    }
  }
  if (Object.getOwnPropertySymbols(samples).length > 0) {
    throw new Error('Array must not have symbol properties');
  }

  const result = [];
  let firstT = null;
  let lastT = null;

  // First pass: validate all points and find time bounds
  for (let i = 0; i < samples.length; i++) {
    const point = samples[i];
    
    // Validate point is plain object
    const proto = Object.getPrototypeOf(point);
    if (proto !== Object.prototype && proto !== null) {
      throw new Error('Points must be plain objects');
    }

    // Get descriptors to validate exact shape
    const pointDesc = Object.getOwnPropertyDescriptors(point);
    const keys = Object.keys(pointDesc);
    
    if (keys.length !== 3 || !keys.includes('x') || !keys.includes('y') || !keys.includes('t')) {
      throw new Error('Points must have exactly x, y, t properties');
    }
    if (pointDesc.x.get || pointDesc.x.set || pointDesc.y.get || pointDesc.y.set || pointDesc.t.get || pointDesc.t.set) {
      throw new Error('Points must not have accessor properties');
    }
    if (Object.getOwnPropertySymbols(point).length > 0) {
      throw new Error('Points must not have symbol properties');
    }

    // Read values
    const { x, y, t } = point;

    // Validate types and finite
    if (typeof x !== 'number' || !Number.isFinite(x)) {
      throw new Error('x must be a finite number');
    }
    if (typeof y !== 'number' || !Number.isFinite(y)) {
      throw new Error('y must be a finite number');
    }
    if (typeof t !== 'number' || !Number.isFinite(t) || t < 0) {
      throw new Error('t must be a finite non-negative number');
    }

    // Validate coordinate bounds
    if (x < 0 || x > 1) {
      throw new Error('x must be within [0, 1]');
    }
    if (y < 0 || y > 1) {
      throw new Error('y must be within [0, 1]');
    }

    // Validate strictly increasing time
    if (i > 0 && t <= lastT) {
      throw new Error('t values must be strictly increasing');
    }

    if (firstT === null) {
      firstT = t;
    }
    lastT = t;
  }

  // Validate time span
  const span = lastT - firstT;
  if (span <= 0 || span > 8) {
    throw new Error('Time span (last - first) must be >0 and <=8 seconds');
  }

  // Second pass: normalize and copy
  for (let i = 0; i < samples.length; i++) {
    const { x, y, t } = samples[i];
    
    let normalizedT;
    if (i === 0) {
      normalizedT = 0;
    } else if (i === samples.length - 1) {
      normalizedT = duration;
    } else {
      normalizedT = ((t - firstT) / span) * duration;
    }

    result.push({ x, y, t: normalizedT });
  }

  // Verify strictly increasing after normalization
  for (let i = 1; i < result.length; i++) {
    if (result[i].t <= result[i - 1].t) {
      throw new Error('Precision collapse: normalized times not strictly increasing');
    }
  }

  return result;
}

/**
 * Converts normalized motion points (t in [0,1]) to trajectory with specified duration.
 * @param {Array} normalizedPoints - Array of 2-128 points with t in [0, 1]
 * @param {number} duration - Target duration in seconds (>0, <=8)
 * @returns {Array} Trajectory with t scaled to [0, duration]
 */
export function motionTrajectory(normalizedPoints, duration) {
  // Validate duration
  if (typeof duration !== 'number' || !Number.isFinite(duration) || duration <= 0 || duration > 8) {
    throw new Error('Duration must be a finite number >0 and <=8');
  }

  // Validate array shape
  if (!Array.isArray(normalizedPoints) || Object.getPrototypeOf(normalizedPoints) !== Array.prototype) {
    throw new Error('Points must be a plain Array');
  }
  if (normalizedPoints.length < 2 || normalizedPoints.length > 128) {
    throw new Error('Points must contain 2-128 points');
  }

  // Check for dense array
  const desc = Object.getOwnPropertyDescriptors(normalizedPoints);
  for (let i = 0; i < normalizedPoints.length; i++) {
    if (!(i in desc) || desc[i].get || desc[i].set) {
      throw new Error('Array must be dense with no accessor properties');
    }
  }
  for (const key of Object.keys(normalizedPoints)) {
    if (key !== 'length' && (Number(key) < 0 || Number(key) >= normalizedPoints.length || !Number.isInteger(Number(key)))) {
      throw new Error('Array must have only numeric indices and length');
    }
  }
  if (Object.getOwnPropertySymbols(normalizedPoints).length > 0) {
    throw new Error('Array must not have symbol properties');
  }

  const result = [];

  // Validate and transform
  for (let i = 0; i < normalizedPoints.length; i++) {
    const point = normalizedPoints[i];
    
    // Validate point is plain object
    const proto = Object.getPrototypeOf(point);
    if (proto !== Object.prototype && proto !== null) {
      throw new Error('Points must be plain objects');
    }

    // Get descriptors
    const pointDesc = Object.getOwnPropertyDescriptors(point);
    const keys = Object.keys(pointDesc);
    
    if (keys.length !== 3 || !keys.includes('x') || !keys.includes('y') || !keys.includes('t')) {
      throw new Error('Points must have exactly x, y, t properties');
    }
    if (pointDesc.x.get || pointDesc.x.set || pointDesc.y.get || pointDesc.y.set || pointDesc.t.get || pointDesc.t.set) {
      throw new Error('Points must not have accessor properties');
    }
    if (Object.getOwnPropertySymbols(point).length > 0) {
      throw new Error('Points must not have symbol properties');
    }

    // Read values
    const { x, y, t } = point;

    // Validate types and finite
    if (typeof x !== 'number' || !Number.isFinite(x)) {
      throw new Error('x must be a finite number');
    }
    if (typeof y !== 'number' || !Number.isFinite(y)) {
      throw new Error('y must be a finite number');
    }
    if (typeof t !== 'number' || !Number.isFinite(t)) {
      throw new Error('t must be a finite number');
    }

    // Validate coordinate bounds
    if (x < 0 || x > 1) {
      throw new Error('x must be within [0, 1]');
    }
    if (y < 0 || y > 1) {
      throw new Error('y must be within [0, 1]');
    }

    // Validate exact endpoints and strictly increasing
    if (i === 0) {
      if (t !== 0) {
        throw new Error('First point t must be exactly 0');
      }
    } else if (i === normalizedPoints.length - 1) {
      if (t !== 1) {
        throw new Error('Last point t must be exactly 1');
      }
      if (t <= normalizedPoints[i - 1].t) {
        throw new Error('t values must be strictly increasing');
      }
    } else {
      if (t <= normalizedPoints[i - 1].t) {
        throw new Error('t values must be strictly increasing');
      }
    }

    // Scale time
    const scaledT = i === normalizedPoints.length - 1 ? duration : t * duration;
    
    result.push({ x, y, t: scaledT });
  }

  // Verify strictly increasing after scaling
  for (let i = 1; i < result.length; i++) {
    if (result[i].t <= result[i - 1].t) {
      throw new Error('Precision collapse: scaled times not strictly increasing');
    }
  }

  return result;
}
```

```javascript
import { test } from 'node:test';
import { strict as assert } from 'node:assert';
import { normalizeMotion, motionTrajectory } from './custom-motion.mjs';

test('normalizeMotion: basic nonuniform timing normalization', () => {
  const samples = [
    { x: 0.1, y: 0.2, t: 1.0 },
    { x: 0.3, y: 0.4, t: 2.5 },
    { x: 0.7, y: 0.8, t: 4.0 }
  ];
  const result = normalizeMotion(samples, 3);
  
  assert.equal(result.length, 3);
  assert.equal(result[0].t, 0);
  assert.equal(result[2].t, 3);
  assert.ok(result[1].t > 0 && result[1].t < 3);
  assert.ok(result[0].t < result[1].t && result[1].t < result[2].t);
  assert.equal(result[0].x, 0.1);
  assert.equal(result[1].y, 0.4);
});

test('normalizeMotion: exact endpoint preservation', () => {
  const samples = [
    { x: 0, y: 0, t: 5.5 },
    { x: 1, y: 1, t: 7.5 }
  ];
  const result = normalizeMotion(samples, 2);
  
  assert.equal(result[0].t, 0);
  assert.equal(result[1].t, 2);
  assert.equal(result[0].x, 0);
  assert.equal(result[1].y, 1);
});

test('normalizeMotion: stationary dwell with repeated coordinates', () => {
  const samples = [
    { x: 0.5, y: 0.5, t: 0 },
    { x: 0.5, y: 0.5, t: 2 },
    { x: 0.5, y: 0.5, t: 4 }
  ];
  const result = normalizeMotion(samples, 4);
  
  assert.equal(result.length, 3);
  assert.equal(result[0].x, 0.5);
  assert.equal(result[1].x, 0.5);
  assert.equal(result[2].x, 0.5);
  assert.ok(result[0].t < result[1].t && result[1].t < result[2].t);
});

test('normalizeMotion: owned copy not mutation', () => {
  const samples = [
    { x: 0.2, y: 0.3, t: 1 },
    { x: 0.8, y: 0.9, t: 3 }
  ];
  const result = normalizeMotion(samples, 2);
  
  assert.notStrictEqual(result, samples);
  assert.notStrictEqual(result[0], samples[0]);
  assert.equal(samples[0].t, 1); // Original unchanged
});

test('normalizeMotion: accepts exactly 128 points', () => {
  const samples = Array.from({ length: 128 }, (_, i) => ({
    x: 0.5, y: 0.5, t: i * 0.01
  }));
  const result = normalizeMotion(samples, 1);
  assert.equal(result.length, 128);
});

test('normalizeMotion: rejects 129 points', () => {
  const samples = Array.from({ length: 129 }, (_, i) => ({
    x: 0.5, y: 0.5, t: i * 0.01
  }));
  assert.throws(() => normalizeMotion(samples, 1), /2-128 points/);
});

test('normalizeMotion: rejects non-finite coordinates', () => {
  assert.throws(() => normalizeMotion([
    { x: NaN, y: 0.5, t: 0 },
    { x: 0.5, y: 0.5, t: 1 }
  ], 1), /finite/);
  
  assert.throws(() => normalizeMotion([
    { x: 0.5, y: Infinity, t: 0 },
    { x: 0.5, y: 0.5, t: 1 }
  ], 1), /finite/);
});

test('normalizeMotion: rejects invalid duration', () => {
  const samples = [{ x: 0, y: 0, t: 0 }, { x: 1, y: 1, t: 1 }];
  
  assert.throws(() => normalizeMotion(samples, 0), /Duration/);
  assert.throws(() => normalizeMotion(samples, -1), /Duration/);
  assert.throws(() => normalizeMotion(samples, 9), /Duration/);
  assert.throws(() => normalizeMotion(samples, NaN), /Duration/);
});

test('normalizeMotion: rejects non-strictly increasing time', () => {
  assert.throws(() => normalizeMotion([
    { x: 0, y: 0, t: 1 },
    { x: 1, y: 1, t: 1 }
  ], 1), /strictly increasing/);
  
  assert.throws(() => normalizeMotion([
    { x: 0, y: 0, t: 2 },
    { x: 1, y: 1, t: 1 }
  ], 1), /strictly increasing/);
});

test('normalizeMotion: rejects invalid time span', () => {
  assert.throws(() => normalizeMotion([
    { x: 0, y: 0, t: 0 },
    { x: 1, y: 1, t: 9 }
  ], 1), /span/);
});

test('normalizeMotion: accessor not invoked', () => {
  let accessed = false;
  const samples = [
    { x: 0, y: 0, get t() { accessed = true; return 0; } },
    { x: 1, y: 1, t: 1 }
  ];
  
  assert.throws(() => normalizeMotion(samples, 1), /accessor/);
  assert.equal(accessed, false);
});

test('normalizeMotion: rejects extra fields', () => {
  assert.throws(() => normalizeMotion([
    { x: 0, y: 0, t: 0, extra: 'field' },
    { x: 1, y: 1, t: 1 }
  ], 1), /exactly x, y, t/);
});

test('normalizeMotion: rejects symbol properties on point', () => {
  const sym = Symbol('test');
  const point = { x: 0, y: 0, t: 0 };
  point[sym] = 'value';
  
  assert.throws(() => normalizeMotion([point, { x: 1, y: 1, t: 1 }], 1), /symbol/);
});

test('normalizeMotion: rejects array with holes', () => {
  const samples = [{ x: 0, y: 0, t: 0 }, { x: 1, y: 1, t: 1 }];
  samples.length = 3;
  
  assert.throws(() => normalizeMotion(samples, 1), /dense/);
});

test('normalizeMotion: rejects array with symbol properties', () => {
  const samples = [{ x: 0, y: 0, t: 0 }, { x: 1, y: 1, t: 1 }];
  samples[Symbol('test')] = 'value';
  
  assert.throws(() => normalizeMotion(samples, 1), /symbol/);
});

test('normalizeMotion: precision collapse with very small duration', () => {
  const samples = [
    { x: 0, y: 0, t: 0 },
    { x: 0.5, y: 0.5, t: 1 },
    { x: 1, y: 1, t: 2 }
  ];
  
  assert.throws(() => normalizeMotion(samples, 1e-16), /Precision collapse/);
});

test('motionTrajectory: retimes normalized path', () => {
  const normalized = [
    { x: 0.1, y: 0.2, t: 0 },
    { x: 0.5, y: 0.5, t: 0.5 },
    { x: 0.9, y: 0.8, t: 1 }
  ];
  const result = motionTrajectory(normalized, 4);
  
  assert.equal(result[0].t, 0);
  assert.equal(result[1].t, 2);
  assert.equal(result[2].t, 4);
  assert.equal(result[0].x, 0.1);
});

test('motionTrajectory: rejects bad first endpoint', () => {
  const bad = [
    { x: 0, y: 0, t: 0.1 },
    { x: 1, y: 1, t: 1 }
  ];
  
  assert.throws(() => motionTrajectory(bad, 2), /First point.*exactly 0/);
});

test('motionTrajectory: rejects bad last endpoint', () => {
  const bad = [
    { x: 0, y: 0, t: 0 },
    { x: 1, y: 1, t: 0.9 }
  ];
  
  assert.throws(() => motionTrajectory(bad, 2), /Last point.*exactly 1/);
});

test('motionTrajectory: rejects non-strictly increasing', () => {
  const bad = [
    { x: 0, y: 0, t: 0 },
    { x: 0.5, y: 0.5, t: 0.3 },
    { x: 1, y: 1, t: 0.3 },
  ];
  
  assert.throws(() => motionTrajectory(bad, 2), /strictly increasing/);
});

test('integration: normalize then retime preserves trajectory shape', () => {
  const samples = [
    { x: 0, y: 0, t: 2 },
    { x: 0.5, y: 1, t: 5 },
    { x: 1, y: 0, t: 6 }
  ];
  
  const normalized = normalizeMotion(samples, 1);
  const trajectory = motionTrajectory(normalized, 3);
  
  assert.equal(trajectory[0].x, 0);
  assert.equal(trajectory[1].x, 0.5);
  assert.equal(trajectory[2].x, 1);
  assert.equal(trajectory[0].t, 0);
  assert.equal(trajectory[2].t, 3);
});
```
